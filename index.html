Quick Start

#include "loading_screen.h"

int main() {
  ls_init(); // Setup terminal

  ls_config_t config = ls_get_default_config();
  config.label = "Downloading assets";
  config.type = LS_TYPE_SMOOTH;

  ls_display(&config, 100);

  ls_cleanup(); // Restore terminal
  return 0;
}
